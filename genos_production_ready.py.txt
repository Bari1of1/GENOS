def validate_env():
    """Validate all required environment variables"""
    required = ["SUPABASE_URL", "SUPABASE_KEY", "CLAUDE_API_KEY"]
    missing = [var for var in required if not os.getenv(var)]
    
    if missing:
        raise ValueError(f"Missing env vars: {', '.join(missing)}\nCreate .env file with:\n" + 
                        "\n".join([f"{var}=your_value" for var in required]))
    
    return {
        "SUPABASE_URL": os.getenv("SUPABASE_URL"),
        "SUPABASE_KEY": os.getenv("SUPABASE_KEY"),
        "CLAUDE_API_KEY": os.getenv("CLAUDE_API_KEY"),
        "JWT_SECRET": os.getenv("JWT_SECRET", hashlib.sha256(os.getenv("SUPABASE_KEY").encode()).hexdigest())
    }

class ProjectCreate(BaseModel):
    name: str
    description: str
    type: str
    language: str
    frameworks: List[str] = []
    
    @validator('name')
    def name_not_empty(cls, v):
        if not v or len(v) < 1 or len(v) > 100:
            raise ValueError('Name must be 1-100 characters')
        return v.strip()
    
    @validator('description')
    def desc_length(cls, v):
        if len(v) > 500:
            raise ValueError('Description max 500 characters')
        return v.strip()
    
    @validator('type')
    def valid_type(cls, v):
        valid = ['api', 'web', 'cli', 'desktop', 'static']
        if v not in valid:
            raise ValueError(f'Type must be one of: {valid}')
        return v
    
    @validator('language')
    def valid_language(cls, v):
        valid = ['python', 'javascript', 'typescript', 'golang']
        if v not in valid:
            raise ValueError(f'Language must be one of: {valid}')
        return v


class DeployRequest(BaseModel):
    type: str
    config: Optional[Dict] = None
    
    @validator('type')
    def valid_deploy_type(cls, v):
        valid = ['docker', 'netlify', 'vercel', 'aws', 'local']
        if v not in valid:
            raise ValueError(f'Deploy type must be one of: {valid}')
        return v

class DatabaseManager:
    def __init__(self, url: str, key: str):
        self.client: Client = create_client(url, key)
    
    def save_project(self, project_id: str, data: dict) -> dict:
        """Save project to database"""
        try:
            result = self.client.table("projects").insert({
                "id": project_id,
                "name": data.get("name"),
                "description": data.get("description"),
                "type": data.get("type"),
                "language": data.get("language"),
                "frameworks": data.get("frameworks", []),
                "status": "active",
                "created_at": datetime.now().isoformat()
            }).execute()
            return {"success": True, "data": result.data}
        except Exception as e:
            return {"success": False, "error": str(e)}
    
    def get_projects(self, user_id: str = None) -> dict:
        """Get projects"""
        try:
            query = self.client.table("projects").select("*")
            if user_id:
                query = query.eq("user_id", user_id)
            result = query.execute()
            return {"success": True, "data": result.data}
        except Exception as e:
            return {"success": False, "error": str(e)}
    
    def save_deployment(self, deployment: dict) -> dict:
        """Save deployment record"""
        try:
            result = self.client.table("deployments").insert({
                "id": str(uuid.uuid4()),
                "project_id": deployment.get("project_id"),
                "type": deployment.get("type"),
                "status": deployment.get("status"),
                "container_id": deployment.get("container_id"),
                "ports": deployment.get("ports"),
                "deployed_at": datetime.now().isoformat()
            }).execute()
            return {"success": True, "data": result.data}
        except Exception as e:
            return {"success": False, "error": str(e)}
    
    def save_upload(self, upload: dict) -> dict:
        """Save file upload record"""
        try:
            result = self.client.table("uploads").insert({
                "id": str(uuid.uuid4()),
                "project_id": upload.get("project_id"),
                "filename": upload.get("filename"),
                "file_type": upload.get("file_type"),
                "file_size": upload.get("file_size"),
                "storage_path": upload.get("storage_path"),
                "uploaded_at": datetime.now().isoformat()
            }).execute()
            return {"success": True, "data": result.data}
        except Exception as e:
            return {"success": False, "error": str(e)}
    
    def save_log(self, project_id: str, level: str, message: str, metadata: dict = None) -> dict:
        """Save log entry"""
        try:
            result = self.client.table("logs").insert({
                "id": str(uuid.uuid4()),
                "project_id": project_id,
                "level": level,
                "message": message,
                "metadata": metadata or {},
                "created_at": datetime.now().isoformat()
            }).execute()
            return {"success": True}
        except Exception as e:
            return {"success": False, "error": str(e)}

class SandboxManager:
    def __init__(self, base_dir: str = "./genos_projects"):
        self.base_dir = Path(base_dir)
        self.base_dir.mkdir(exist_ok=True)
    
    def create_sandbox(self, project_id: str) -> dict:
        """Create project sandbox"""
        try:
            project_path = self.base_dir / project_id
            project_path.mkdir(exist_ok=True)
            
            for dir in ["src", "tests", "docs", ".github/workflows"]:
                (project_path / dir).mkdir(parents=True, exist_ok=True)
            
            return {"success": True, "path": str(project_path)}
        except Exception as e:
            return {"success": False, "error": str(e)}
    
    def write_file(self, project_id: str, filepath: str, content: str) -> dict:
        """Write file to sandbox"""
        try:
            project_path = self.base_dir / project_id
            file_path = project_path / filepath
            file_path.parent.mkdir(parents=True, exist_ok=True)
            file_path.write_text(content)
            return {"success": True, "path": str(file_path)}
        except Exception as e:
            return {"success": False, "error": str(e)}
    
    def get_file(self, project_id: str, filepath: str) -> Optional[str]:
        """Read file from sandbox"""
        try:
            file_path = self.base_dir / project_id / filepath
            if file_path.exists():
                return file_path.read_text()
            return None
        except Exception as e:
            return None


class AIGenerator:
    def __init__(self, api_key: str):
        self.client = anthropic.Anthropic(api_key=api_key)
    
    def generate_project(self, spec: dict) -> dict:
        """Generate project files with Claude"""
        prompt = f"""
        Generate a complete, working {spec['language']} {spec['type']} application.
        
        Project: {spec['name']}
        Description: {spec['description']}
        Type: {spec['type']}
        Frameworks: {', '.join(spec.get('frameworks', []))}
        
        Return ONLY valid JSON (no markdown, no code blocks):
        {{
            "files": {{"path/to/file": "content", ...}},
            "dependencies": ["dep1", "dep2"],
            "scripts": {{"start": "...", "build": "..."}},
            "dockerfile": "Dockerfile content",
            "github_actions": "CI workflow"
        }}
        """
        
        try:
            message = self.client.messages.create(
                model="claude-3-5-sonnet-20241022",
                max_tokens=4096,
                messages=[{"role": "user", "content": prompt}]
            )
            
            response_text = message.content[0].text
            json_start = response_text.find('{')
            json_end = response_text.rfind('}') + 1
            
            if json_start == -1 or json_end == 0:
                raise ValueError("No JSON found in response")
            
            return json.loads(response_text[json_start:json_end])
        except json.JSONDecodeError:
            return {"error": "Invalid JSON from AI", "files": {}}
        except Exception as e:
            return {"error": str(e), "files": {}}

@app.post("/api/projects")
def create_project(project: ProjectCreate):
    """Create new project with AI generation"""
    try:
        project_id = f"{project.name.lower().replace(' ', '-')}_{uuid.uuid4().hex[:8]}"
        
        # Create sandbox
        sandbox_result = sandbox.create_sandbox(project_id)
        if not sandbox_result["success"]:
            raise HTTPException(status_code=500, detail=sandbox_result["error"])
        
        # Generate files with AI
        spec = project.dict()
        ai_output = ai.generate_project(spec)
        
        if "error" in ai_output:
            db.save_log(project_id, "error", f"AI generation failed: {ai_output['error']}")
            raise HTTPException(status_code=500, detail=ai_output["error"])
        
        # Write files
        for filepath, content in ai_output.get("files", {}).items():
            sandbox.write_file(project_id, filepath, content)
        
        if ai_output.get("dockerfile"):
            sandbox.write_file(project_id, "Dockerfile", ai_output["dockerfile"])
        
        # Save to database
        db_result = db.save_project(project_id, spec)
        db.save_log(project_id, "info", f"Project created: {project.name}")
        
        return {
            "success": True,
            "project_id": project_id,
            "name": project.name,
            "files": list(ai_output.get("files", {}).keys()),
            "dependencies": ai_output.get("dependencies", [])
        }
    
    except HTTPException:
        raise
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))

@app.post("/api/projects/{project_id}/upload")
async def upload_file(project_id: str, file: UploadFile = File(...)):
    """Upload file to project"""
    try:
        content = await file.read()
        file_size = len(content)
        
        # Save file
        project_path = projects_dir / project_id
        project_path.mkdir(parents=True, exist_ok=True)
        file_path = project_path / file.filename
        
        with open(file_path, "wb") as f:
            f.write(content)
        
        # Save to database
        db.save_upload({
            "project_id": project_id,
            "filename": file.filename,
            "file_type": file.content_type,
            "file_size": file_size,
            "storage_path": str(file_path)
        })
        
        db.save_log(project_id, "info", f"File uploaded: {file.filename}")
        
        return {
            "success": True,
            "filename": file.filename,
            "size": file_size,
            "path": f"/projects/{project_id}/{file.filename}"
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))


@app.post("/api/projects/{project_id}/deploy")
def deploy_project(project_id: str, deploy_req: DeployRequest):
    """Deploy project"""
    try:
        db.save_log(project_id, "info", f"Deployment started: {deploy_req.type}")
        
        if deploy_req.type == "docker":
            project_path = projects_dir / project_id
            
            # Check Dockerfile
            if not (project_path / "Dockerfile").exists():
                raise HTTPException(status_code=400, detail="No Dockerfile found")
            
            # Build
            build_result = subprocess.run(
                ["docker", "build", "-t", f"genos-{project_id}", "."],
                cwd=str(project_path),
                capture_output=True,
                text=True,
                timeout=300
            )
            
            if build_result.returncode != 0:
                db.save_log(project_id, "error", f"Docker build failed: {build_result.stderr}")
                raise HTTPException(status_code=500, detail="Docker build failed")
            
            # Run
            run_result = subprocess.run(
                ["docker", "run", "-d", "-p", "8000:8000", "--name", project_id, f"genos-{project_id}"],
                capture_output=True,
                text=True
            )
            
            if run_result.returncode == 0:
                db.save_deployment({
                    "project_id": project_id,
                    "type": "docker",
                    "status": "running",
                    "container_id": run_result.stdout.strip(),
                    "ports": {"8000": "8000"}
                })
                db.save_log(project_id, "info", "Docker deployment successful")
                return {"success": True, "status": "running", "container": run_result.stdout.strip()}
            else:
                db.save_log(project_id, "error", f"Docker run failed: {run_result.stderr}")
                raise HTTPException(status_code=500, detail="Docker run failed")
        
        return {"success": True, "type": deploy_req.type}
    
    except HTTPException:
        raise
    except Exception as e:
        db.save_log(project_id, "error", str(e))
        raise HTTPException(status_code=500, detail=str(e))

@app.delete("/api/projects/{project_id}")
def delete_project(project_id: str):
    """Delete project"""
    try:
        # Delete files
        project_path = projects_dir / project_id
        if project_path.exists():
            shutil.rmtree(project_path)
        
        # Delete from database
        db.client.table("projects").delete().eq("id", project_id).execute()
        
        db.save_log(project_id, "info", "Project deleted")
        return {"success": True}
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
